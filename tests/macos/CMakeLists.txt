add_executable(wui_macos_test integration.mm)
set_target_properties(wui_macos_test PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
target_compile_options(wui_macos_test PRIVATE -fobjc-arc)
target_compile_definitions(wui_macos_test PRIVATE WUI_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
target_link_libraries(wui_macos_test PRIVATE wui)
add_test(NAME macos_integration COMMAND wui_macos_test)
set_tests_properties(macos_integration PROPERTIES TIMEOUT 30)

foreach(example IN ITEMS hello_world simple demo)
    set(example_dir "${PROJECT_SOURCE_DIR}/examples/${example}")
    if(example STREQUAL "hello_world")
        set(entry hw.cpp)
    elseif(example STREQUAL "simple")
        set(entry simple.cpp)
    else()
        set(entry demo.cpp)
    endif()
    file(GLOB_RECURSE example_sources CONFIGURE_DEPENDS "${example_dir}/*.cpp")
    list(REMOVE_ITEM example_sources "${example_dir}/${entry}")
    add_executable(wui_${example}_smoke MACOSX_BUNDLE example_smoke.mm ${example_sources})
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${example}-work")
    target_compile_definitions(wui_${example}_smoke PRIVATE
        WUI_TEST_WORK_DIR="${CMAKE_CURRENT_BINARY_DIR}/${example}-work"
        WUI_EXAMPLE_FILE="${example_dir}/${entry}" WUI_SCREENSHOT="/tmp/wui-macos-${example}.png")
    target_include_directories(wui_${example}_smoke PRIVATE "${example_dir}")
    target_compile_options(wui_${example}_smoke PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    set_target_properties(wui_${example}_smoke PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    target_link_libraries(wui_${example}_smoke PRIVATE wui)
    add_custom_command(TARGET wui_${example}_smoke POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${example_dir}/res" "$<TARGET_BUNDLE_CONTENT_DIR:wui_${example}_smoke>/Resources/res")
    add_test(NAME macos_${example} COMMAND wui_${example}_smoke)
    set_tests_properties(macos_${example} PROPERTIES TIMEOUT 15)
endforeach()
